df = read.csv("data/SeoulBikeData.csv", fileEncoding = "latin1") %>% clean_names()
trialdf1 = df %>% select(-functioning_day, -holiday) %>% group_by(hour, seasons) %>% 
  summarise(
    Mean_Div_100 = mean(rented_bike_count, na.rm = TRUE)/100
  )
## `summarise()` has regrouped the output.
## ℹ Summaries were computed grouped by hour and seasons.
## ℹ Output is grouped by hour.
## ℹ Use `summarise(.groups = "drop_last")` to silence this message.
## ℹ Use `summarise(.by = c(hour, seasons))` for per-operation grouping
##   (`?dplyr::dplyr_by`) instead.
trialdf2 = df %>% select(-functioning_day, -holiday) %>% group_by(hour, seasons) %>% 
  summarise(
    m = mean(temperature_c, na.rm = TRUE)
  )
## `summarise()` has regrouped the output.
## ℹ Summaries were computed grouped by hour and seasons.
## ℹ Output is grouped by hour.
## ℹ Use `summarise(.groups = "drop_last")` to silence this message.
## ℹ Use `summarise(.by = c(hour, seasons))` for per-operation grouping
##   (`?dplyr::dplyr_by`) instead.
ggplot() +
  geom_bar(data = trialdf1, aes(x = hour, y = Mean_Div_100, fill = "Mean Rented Bike Count (/100)"), stat = "identity", alpha = 0.7) +
  geom_point(data = trialdf2, aes(x = hour, y = m, colour = "Mean Temperature (C)"), size = 2) +
  facet_wrap(~ seasons) +
  scale_fill_manual(name = NULL , values = c("Mean Rented Bike Count (/100)" = "lightblue")) +
  scale_color_manual(name = "Key", values = c("Mean Temperature (C)" = "grey")) +
  labs(
    x = "Hour",
    y = "Mean Rented Bike Count (/100), Mean Temperature (C)",
    title = "Effect Temperature has on Rented Bike Count"
  )

  summarise(
    mean_bike_count = mean(rented_bike_count, na.rm = TRUE)
  )
## `summarise()` has regrouped the output.
## ℹ Summaries were computed grouped by hour and seasons.

trialdf4 = df %>% select(-functioning_day, -holiday) %>% group_by(hour, seasons) %>% 
  summarise(
    mean_temperature = mean(temperature_c, na.rm = TRUE)
  )
## `summarise()` has regrouped the output.
## ℹ Summaries were computed grouped by hour and seasons.
## ℹ Output is grouped by hour.

trialdf5 <- inner_join(trialdf3, trialdf4, by = c("seasons", "hour")) 
ggplot(data = trialdf5, 
       mapping = aes(x = mean_bike_count, y = mean_temperature, colour = seasons)) +
geom_point() + 
  facet_wrap(~ seasons) +
labs(title = "Correlation Between Temperature and Bike Demand",
       x = "Mean Number of Bikes Demanded", 
       y = "Mean Temperature (°C)",
